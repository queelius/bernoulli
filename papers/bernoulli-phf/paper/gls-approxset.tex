\newglossaryentry{gls-approxset}
{
    name={approximate set},
    description={A set with random errors in membership queries, parameterized by false positive and false negative rates.}
}

\newglossaryentry{gls-fprate}
{
    name={false positive rate},
    description={The probability that a non-member of the objective set is incorrectly reported as a member.}
}
