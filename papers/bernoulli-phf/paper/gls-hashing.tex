\newglossaryentry{gls-phf}
{
    name={Perfect Hash Filter},
    description={A data structure that implements the positive approximate set abstract data type using a perfect hash function and hash fingerprints.}
}

\newglossaryentry{gls-ph}
{
    name={perfect hash function},
    description={A hash function that is injective on a known set $S$, mapping each element to a unique index.}
}

\newglossaryentry{gls-blf}
{
    name={Bloom filter},
    description={A probabilistic data structure that implements the positive approximate set abstract data type using multiple hash functions and a bit array.}
}

\newglossaryentry{gls-pmf}
{
    name={Perfect Map Filter},
    description={A data structure that implements the approximate map abstract data type using a perfect hash function and hash fingerprints for both keys and values.}
}
